Stdout capture pattern for REQ-F-009 test:
```go
func TestInitProvider_StdoutExporter_WritesToStderr(t *testing.T) {
    oldStdout := os.Stdout
    r, w, _ := os.Pipe()
    os.Stdout = w

    cfg := config.ObservabilityConfig{Enabled: true, Exporter: "stdout", ServiceName: "test"}
    shutdown, err := observability.InitProvider(cfg)
    require.NoError(t, err)
    ctx, span := otel.Tracer("test").Start(context.Background(), "span")
    span.End()
    _ = ctx
    require.NoError(t, shutdown(context.Background()))

    w.Close()
    os.Stdout = oldStdout
    var buf bytes.Buffer
    io.Copy(&buf, r)
    assert.Equal(t, 0, buf.Len(), "expected zero bytes on stdout")
    t.Cleanup(func() { observability.NoopProvider() })
}
```
