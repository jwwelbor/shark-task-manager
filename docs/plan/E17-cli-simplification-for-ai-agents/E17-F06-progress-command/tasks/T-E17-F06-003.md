### Technical Approach

#### Test file structure

```go
// File: internal/cli/commands/progress_test.go
package commands

import (
    "bytes"
    "encoding/json"
    "io"
    "os"
    "testing"

    "github.com/jwwelbor/shark-task-manager/internal/status"
    "github.com/spf13/cobra"
)

// newTestProgressCmd creates a minimal cobra.Command with all progress flags registered.
func newTestProgressCmd() *cobra.Command {
    cmd := &cobra.Command{}
    cmd.Flags().String("epic", "", "")
    cmd.Flags().String("recent", "", "")
    cmd.Flags().Bool("include-archived", false, "")
    return cmd
}

// captureStdout captures standard output during fn() and returns it as a string.
func captureStdout(fn func()) string {
    old := os.Stdout
    r, w, _ := os.Pipe()
    os.Stdout = w
    fn()
    _ = w.Close()
    os.Stdout = old
    var buf bytes.Buffer
    _, _ = io.Copy(&buf, r)
    return buf.String()
}
```

#### Table-driven test pattern for argument parsing

```go
func TestParseProgressRequest(t *testing.T) {
    tests := []struct {
        name            string
        args            []string
        epicFlag        string
        recentFlag      string
        includeArchived bool
        wantEpicKey     string
        wantRecent      string
        wantArchived    bool
        wantErr         bool
    }{
        // TC-PARSE-01
        {name: "no args no flags", args: []string{}, wantEpicKey: "", wantErr: false},
        // TC-PARSE-02
        {name: "epic positional", args: []string{"E05"}, wantEpicKey: "E05", wantErr: false},
        // TC-PARSE-03
        {name: "epic flag", args: []string{}, epicFlag: "E05", wantEpicKey: "E05", wantErr: false},
        // TC-PARSE-04
        {name: "positional overrides flag", args: []string{"E05"}, epicFlag: "E07", wantEpicKey: "E05", wantErr: false},
        // TC-PARSE-05
        {name: "combined feature format", args: []string{"E05-F02"}, wantEpicKey: "E05", wantErr: false},
        // TC-PARSE-06
        {name: "epic and feature positional", args: []string{"E05", "F02"}, wantEpicKey: "E05", wantErr: false},
        // TC-PARSE-07
        {name: "recent window flag", args: []string{}, recentFlag: "7d", wantRecent: "7d", wantErr: false},
        // TC-PARSE-08
        {name: "include archived flag", args: []string{}, includeArchived: true, wantArchived: true, wantErr: false},
        // TC-PARSE-09
        {name: "too many args", args: []string{"E05", "F02", "extra"}, wantErr: true},
        // TC-PARSE-11
        {name: "all flags combined", args: []string{"E07"}, recentFlag: "30d", includeArchived: true,
         wantEpicKey: "E07", wantRecent: "30d", wantArchived: true, wantErr: false},
        // TC-PARSE-12
        {name: "empty epic flag", args: []string{}, epicFlag: "", wantEpicKey: "", wantErr: false},
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            cmd := newTestProgressCmd()
            if tt.epicFlag != "" {
                _ = cmd.Flags().Set("epic", tt.epicFlag)
            }
            if tt.recentFlag != "" {
                _ = cmd.Flags().Set("recent", tt.recentFlag)
            }
            if tt.includeArchived {
                _ = cmd.Flags().Set("include-archived", "true")
            }

            req, err := parseProgressRequest(cmd, tt.args)
            if tt.wantErr {
                if err == nil {
                    t.Error("expected error, got nil")
                }
                return
            }
            if err != nil {
                t.Fatalf("unexpected error: %v", err)
            }
            if req.EpicKey != tt.wantEpicKey {
                t.Errorf("EpicKey: got %q, want %q", req.EpicKey, tt.wantEpicKey)
            }
            if req.RecentWindow != tt.wantRecent {
                t.Errorf("RecentWindow: got %q, want %q", req.RecentWindow, tt.wantRecent)
            }
            if req.IncludeArchived != tt.wantArchived {
                t.Errorf("IncludeArchived: got %v, want %v", req.IncludeArchived, tt.wantArchived)
            }
        })
    }
}
```

#### JSON output tests

Build a minimal `status.StatusDashboard` struct directly -- no service call needed.

```go
func TestOutputProgressJSON_MinimalDashboard(t *testing.T) {
    dashboard := &status.StatusDashboard{
        Summary: &status.ProjectSummary{
            OverallProgress: 0.0,
            Epics:           &status.CountBreakdown{},
            Features:        &status.CountBreakdown{},
            Tasks:           &status.StatusBreakdown{},
        },
        Epics:        []*status.EpicSummary{},
        ActiveTasks:  map[string][]*status.TaskInfo{},
        BlockedTasks: []*status.BlockedTaskInfo{},
    }

    output := captureStdout(func() {
        if err := outputProgressJSON(dashboard); err != nil {
            t.Errorf("outputProgressJSON returned error: %v", err)
        }
    })

    var parsed map[string]interface{}
    if err := json.Unmarshal([]byte(output), &parsed); err != nil {
        t.Fatalf("output is not valid JSON: %v\nOutput: %s", err, output)
    }

    for _, key := range []string{"summary", "epics", "active_tasks", "blocked_tasks"} {
        if _, ok := parsed[key]; !ok {
            t.Errorf("missing key %q in JSON output", key)
        }
    }
}

func TestOutputProgressJSON_ProgressValue(t *testing.T) {
    dashboard := &status.StatusDashboard{

        _ = outputProgressJSON(dashboard)
```
