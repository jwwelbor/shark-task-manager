## Implementation

### Pattern
```go
func runEpicGet(cmd *cobra.Command, args []string) error {
    epicKey := args[0]

    svc := cli.GetEpicService()
    epic, err := svc.GetEpic(cmd.Context(), epicKey)
    if err != nil {
        return err
    }

    if cli.GlobalConfig.JSON {
        return cli.OutputJSON(epic)
    }
    // Format table output
    return nil
}
```
