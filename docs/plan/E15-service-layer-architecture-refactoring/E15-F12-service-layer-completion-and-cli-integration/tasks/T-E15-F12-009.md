## Implementation

### Pattern
```go
func runFeatureGet(cmd *cobra.Command, args []string) error {
    featureKey := args[0]

    svc := cli.GetFeatureService()
    feature, err := svc.GetFeature(cmd.Context(), featureKey)
    if err != nil {
        return err
    }

    if cli.GlobalConfig.JSON {
        return cli.OutputJSON(feature)
    }
    // Format table output
    return nil
}
```
