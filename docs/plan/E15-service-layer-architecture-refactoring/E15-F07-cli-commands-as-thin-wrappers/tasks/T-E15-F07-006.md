### Centralized Error Handler

Add to `internal/cli/commands/helpers.go` (or create if missing):

```go
// handleServiceError translates service-layer errors to CLI exit codes.
// Exits the process with the appropriate code for the error type.
// Exit code 1: NotFoundError (entity does not exist)
// Exit code 2: All other errors (db, system)
// Exit code 3: Workflow/transition errors (invalid state)
func handleServiceError(err error, entityKey string) {
    if err == nil {
        return
    }

    var notFoundErr *repository.NotFoundError
    if errors.As(err, &notFoundErr) {
        cli.Error(fmt.Sprintf("%s not found: %s", notFoundErr.Entity, notFoundErr.Key))
        os.Exit(1)
    }

    var workflowErr *workflow.TransitionError
    if errors.As(err, &workflowErr) {
        cli.Error(fmt.Sprintf("Invalid operation: %v", err))
        os.Exit(3)
    }

    cli.Error(fmt.Sprintf("Error: %v", err))
    os.Exit(2)
}
```
