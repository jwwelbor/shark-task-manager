### Technical Approach

**Test Structure**: Follow project patterns from `.claude/rules/services/testing.md`
- Use mock repositories (never real database in service tests)
- Table-driven tests for parameter combinations
- Subtests for related checks (`t.Run("subtest name", ...)`)
- Clear test names describing scenario and expected outcome

**Mock Pattern**:
```go
func TestTaskService_ListTasksPagination(t *testing.T) {
    mockRepo := &MockTaskRepository{
        ListFunc: func(ctx context.Context) ([]*models.Task, error) {
            // Return 100 tasks for pagination testing
            tasks := make([]*models.Task, 100)
            for i := 0; i < 100; i++ {
                tasks[i] = &models.Task{
                    Key:    fmt.Sprintf("E15-F04-%03d", i+1),
                    Title:  fmt.Sprintf("Task %d", i+1),
                    Status: "todo",
                }
            }
            return tasks, nil
        },
    }

    svc := services.NewTaskService(mockRepo, mockWorkflow, nil, nil)

    tests := []struct {
        name   string
        limit  int
        offset int
        want   int
    }{
        {"first page", 10, 0, 10},
        {"second page", 10, 10, 10},
        {"last page partial", 10, 95, 5},
        {"offset beyond total", 10, 100, 0},
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            filters := services.TaskFilters{
                Limit:  tt.limit,
                Offset: tt.offset,
            }
            tasks, total, err := svc.ListTasks(context.Background(), filters)
            assert.NoError(t, err)
            assert.Equal(t, 100, total) // Total unchanged
            assert.Equal(t, tt.want, len(tasks))
        })
    }
}
```

**Benchmark Pattern**:
```go
func BenchmarkTaskService_DetectCycle(b *testing.B) {
    // Create graph with 1000 nodes
    graph := make(map[string][]string)
    for i := 0; i < 1000; i++ {
        key := fmt.Sprintf("T-%03d", i)
        if i > 0 {
            graph[key] = []string{fmt.Sprintf("T-%03d", i-1)}
        }
    }

    // Add cycle at end
    graph["T-999"] = append(graph["T-999"], "T-000")

    mockRepo := &MockTaskRepository{
        GetTaskDependentsFunc: func(ctx context.Context, taskKey string) ([]*models.Task, error) {
            deps := make([]*models.Task, 0)
            for _, depKey := range graph[taskKey] {
                deps = append(deps, &models.Task{Key: depKey})
            }
            return deps, nil
        },
    }

    svc := services.NewTaskService(mockRepo, mockWorkflow, nil, nil)

    b.ResetTimer()
    for i := 0; i < b.N; i++ {
        _, err := svc.ValidateDependencies(context.Background(), "T-000", "in_progress")
        if err == nil {
            b.Fatal("expected cycle detection error")
        }
    }
}
```
