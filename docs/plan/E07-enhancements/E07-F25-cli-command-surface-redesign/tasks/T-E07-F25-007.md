```go
func TestDetectEntityType(t *testing.T) {
    tests := []struct {
        name     string
        key      string
        expected string
    }{
        // Epic keys
        {"numeric epic", "E07", "epic"},
        {"lowercase epic", "e07", "epic"},
        {"slugged epic", "E07-enhancements", "epic"},

        // Feature keys
        {"full feature", "E07-F01", "feature"},
        {"suffix feature", "F01", "feature"},
        {"slugged feature", "E07-F01-auth", "feature"},
        {"suffix slugged feature", "F01-auth", "feature"},

        // Task keys
        {"short task", "E07-F01-001", "task"},
        {"traditional task", "T-E07-F01-001", "task"},
        {"slugged task", "E07-F01-001-implement-login", "task"},
        {"traditional slugged task", "T-E07-F01-001-implement-login", "task"},

        // Unknown
        {"empty string", "", "unknown"},
        {"random string", "invalid", "unknown"},
        {"numbers only", "12345", "unknown"},
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            result := DetectEntityType(tt.key)
            if result != tt.expected {
                t.Errorf("DetectEntityType(%q) = %q, want %q", tt.key, result, tt.expected)
            }
        })
    }
}
```

### Test 2: Alias Command Registration (in `internal/cli/commands/aliases_test.go`)

```go
func TestAliasCommandsRegistered(t *testing.T) {
    // Verify all alias commands are registered with root
    aliasNames := []string{"next", "start", "done", "block", "unblock"}

    for _, name := range aliasNames {
        t.Run(name, func(t *testing.T) {
            cmd, _, err := cli.RootCmd.Find([]string{name})
            if err != nil {
                t.Errorf("alias %q not found in root command: %v", name, err)
            }
            if cmd == nil {
                t.Errorf("alias %q returned nil command", name)
            }
            if cmd.GroupID != "quick" && cmd.GroupID != "essentials" {
                t.Errorf("alias %q has unexpected GroupID %q", name, cmd.GroupID)
            }
        })
    }
}

func TestAliasNextHasExpectedFlags(t *testing.T) {
    cmd, _, _ := cli.RootCmd.Find([]string{"next"})

        t.Fatal("next command not found")

    for _, flag := range expectedFlags {
```

### Test 3: Create Dispatcher Registration (in `internal/cli/commands/create_test.go`)

```go
func TestCreateDispatcherSubcommands(t *testing.T) {
    // Verify create command has expected subcommands
    createCmd, _, _ := cli.RootCmd.Find([]string{"create"})
    if createCmd == nil {
        t.Fatal("create command not found")
    }

    subcommandNames := []string{"epic", "feature", "task"}
    for _, name := range subcommandNames {
        t.Run(name, func(t *testing.T) {
            subcmd, _, err := createCmd.Find([]string{name})
            if err != nil || subcmd == nil {
                t.Errorf("create subcommand %q not found: %v", name, err)
            }
        })
    }
}

func TestCreateTaskHasExpectedFlags(t *testing.T) {
    createCmd, _, _ := cli.RootCmd.Find([]string{"create"})
    taskCmd, _, _ := createCmd.Find([]string{"task"})
    if taskCmd == nil {
        t.Fatal("create task command not found")
    }

    expectedFlags := []string{"agent", "priority", "epic", "feature", "description"}
    for _, flag := range expectedFlags {
        if taskCmd.Flags().Lookup(flag) == nil {
            t.Errorf("create task missing expected flag: %s", flag)
        }
    }
}
```
