```go
// TestListRelatedTaskKeys_ContextCancellation tests AC-7
func TestListRelatedTaskKeys_ContextCancellation(t *testing.T) {
    ctx := context.Background()
    database := test.GetTestDB()
    db := NewDB(database)
    repo := NewTaskRelationshipRepository(db)
    taskRepo := NewTaskRepository(db)

    // Clean before test
    _, _ = database.ExecContext(ctx, "DELETE FROM tasks WHERE key = 'TEST-CTX-CANCEL'")

    // Seed test data
    epicID, featureID := test.SeedTestData()

    // Create task
    task := &models.Task{
        Key:       "TEST-CTX-CANCEL",
        Title:     "Task",
        Status:    "todo",
        EpicID:    epicID,
        FeatureID: featureID,
    }
    err := taskRepo.Create(ctx, task)
    if err != nil {
        t.Fatalf("Failed to create task: %v", err)
    }
    defer database.ExecContext(ctx, "DELETE FROM tasks WHERE id = ?", task.ID)

    // Create context with 1ms timeout
    ctxTimeout, cancel := context.WithTimeout(ctx, 1*time.Millisecond)
    defer cancel()

    // Wait for timeout
    time.Sleep(2 * time.Millisecond)

    // Execute method with cancelled context
    keys, err := repo.ListRelatedTaskKeys(ctxTimeout, task.ID)

    // Assert
    if err == nil {
        t.Error("expected error for cancelled context, got nil")
    }
    if keys != nil {
        t.Errorf("expected nil keys on error, got: %v", keys)
    }
    if !errors.Is(err, context.DeadlineExceeded) {
        t.Errorf("expected context.DeadlineExceeded error, got: %v", err)
    }
}
```

**Location**: Add to `internal/repository/task_relationship_repository_test.go`
