```go
func GetOrchestratorEngine() *OrchestratorRenderer {
    engineOnce.Do(func() {
        // Use test directory if set, otherwise default
        templateDir := "templates"
        if testTemplateDir != "" {
            templateDir = testTemplateDir
        }

        engineInstance, engineError = NewOrchestratorRenderer(templateDir)
        if engineError != nil {
            panic(fmt.Sprintf("Failed to initialize template engine: %v", engineError))
        }
    })

    return engineInstance
}
```

**2. Template Precompilation (lines 28-58)**
```go
func NewOrchestratorRenderer(templateDir string) (*OrchestratorRenderer, error) {
    pattern := filepath.Join(templateDir, "*", "*.tmpl")

    tmpl := template.New("orchestrator").Funcs(orchestratorFuncs())

    matches, err := filepath.Glob(pattern)
    if err != nil {
        return nil, fmt.Errorf("failed to glob templates: %w", err)
    }

    if len(matches) == 0 {
        return &OrchestratorRenderer{templates: tmpl, templateDir: templateDir}, nil
    }

    tmpl, err = tmpl.ParseGlob(pattern)
    if err != nil {
        return nil, fmt.Errorf("failed to parse templates: %w", err)
    }

    return &OrchestratorRenderer{templates: tmpl, templateDir: templateDir}, nil
}
```

**Strengths:**
- ✅ Gracefully handles empty directories (valid use case)
- ✅ Error wrapping provides context (`"failed to parse templates: %w"`)
- ✅ Pattern matches `templateDir/*/*.tmpl` (subdirectory organization)
- ✅ Custom functions registered before parsing

**Minor Observation:**
- Could add debug logging for number of templates loaded (optional enhancement)

**3. Render Method (lines 84-102)**
```go
func (r *OrchestratorRenderer) Render(templateName string, vars map[string]string) (string, error) {
    baseName := filepath.Base(templateName)

    tmpl := r.templates.Lookup(baseName)
    if tmpl == nil {
        return "", fmt.Errorf("template not found: %s", templateName)
    }

    var buf bytes.Buffer
    if err := tmpl.Execute(&buf, vars); err != nil {
        return "", fmt.Errorf("failed to execute template: %w", err)
    }

    return buf.String(), nil
}
```

**Strengths:**
- ✅ Base name extraction allows flexible path formats
- ✅ Clear error messages (`"template not found"`, `"failed to execute"`)
- ✅ Buffer-based rendering avoids string concatenation
- ✅ Error wrapping maintains context

**4. Custom Functions (lines 104-131)**
```go
func orchestratorFuncs() template.FuncMap {
    return template.FuncMap{
        "eq": func(a, b interface{}) bool { return a == b },
        "ne": func(a, b interface{}) bool { return a != b },
        "isEmpty": func(s string) bool { return strings.TrimSpace(s) == "" },
        "isSimple": func(tier string) bool { return tier == "SIMPLE" },
        "isStandard": func(tier string) bool { return tier == "STANDARD" },
        "isComplex": func(tier string) bool { return tier == "COMPLEX" },
    }
}
```

**Strengths:**
- ✅ Clean, concise implementations
- ✅ Well-named functions matching domain language
- ✅ `isEmpty` properly trims whitespace
- ✅ Tier helpers are case-sensitive (matches config convention)
